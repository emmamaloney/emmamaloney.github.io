/* ---------------- Updated Roster List Renderer ---------------- */
async function renderRosterList() {
  // 1. Correctly target and declare the structural layout DOM reference upfront
  const listElement = document.getElementById('roster-list');
  if (!listElement) return;

  let playerNames = [];

  try {
    // 2. Fetch the unique array from your FastAPI cards_analytics server
    const response = await fetch(`${RENDER_URL}/dataset/players`, {
      method: 'GET'
    });
    
    if (response.ok) {
      // Server returns a flat array of strings, e.g., ["Al", "Camilla", "Casper", "Emma"]
      playerNames = await response.json(); 
    } else {
      throw new Error("Server response failed, falling back to local roster state.");
    }
  } catch (e) {
    console.warn("Fetch failed:", e);
    // Fallback to your local user state array variable if the server is offline
    playerNames = typeof roster !== 'undefined' ? roster : [];
  }
  
  // 3. Render placeholder message if no players exist anywhere
  if (playerNames.length === 0) {
    listElement.innerHTML = '<li style="color:#8a806a;">No saved players yet.</li>';
    return;
  }
  
  listElement.innerHTML = playerNames.map((name, index) => `
    <li>
      ${escapeHtml(name)} 
    </li>
  `).join('');
  // remove roster name: <button onclick="removeRosterName(${index})">Remove</button>
  
}

async function addRosterNameFromModal(){
  const input = document.getElementById('new-roster-name');
  const name = input.value.trim();
  if(!name) return;
  if(!roster.includes(name)){
    roster.push(name);
    await storageSet('roster', roster);
  }
  input.value = '';
  renderRosterList();
}

async function removeRosterName(i){
  roster.splice(i,1);
  await storageSet('roster', roster);
  renderRosterList();
}
